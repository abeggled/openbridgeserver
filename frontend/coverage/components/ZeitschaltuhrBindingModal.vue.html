    const i = cfg.selected_holidays.indexOf(name)
    if (i >= 0) {
      cfg.selected_holidays.splice(i, 1)
    } else {
      cfg.selected_holidays.push(name)
      if (cfg.selected_holidays.length === holidays.value.length) cfg.selected_holidays = []
    }
  }
}
